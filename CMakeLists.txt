cmake_minimum_required(VERSION 3.21)
project(WizPlayer VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_library(binjnes STATIC
  third_party/binjnes/src/emulator.c
  third_party/binjnes/src/common.c)
set_target_properties(binjnes PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
target_include_directories(binjnes PUBLIC third_party/binjnes/src)
if(MSVC)
  target_compile_definitions(binjnes PUBLIC BINJNES_MSVC PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(binjnes PRIVATE /utf-8 /wd4146 /wd4098)
elseif(CMAKE_C_COMPILER_ID MATCHES "Clang")
  target_compile_definitions(binjnes PUBLIC BINJNES_CLANG)
elseif(CMAKE_C_COMPILER_ID STREQUAL "GNU")
  target_compile_definitions(binjnes PUBLIC BINJNES_GCC)
endif()

set(CORE ${CMAKE_CURRENT_SOURCE_DIR}/third_party/gambatte/libgambatte)
file(GLOB_RECURSE CORE_SOURCES CONFIGURE_DEPENDS ${CORE}/src/*.cpp)
add_library(gambatte STATIC ${CORE_SOURCES} ${CORE}/libretro/gambatte_log.c)
set_property(TARGET gambatte PROPERTY CXX_STANDARD 14)
if(MSVC)
  add_compile_options(/utf-8)
  target_compile_options(gambatte PRIVATE /utf-8)
endif()
target_include_directories(gambatte PUBLIC ${CORE}/include PRIVATE ${CORE}/src ${CORE}/libretro ${CORE}/libretro-common/include ${CMAKE_CURRENT_SOURCE_DIR}/third_party/gambatte/common)
target_compile_definitions(gambatte PUBLIC HAVE_STDINT_H HAVE_INTTYPES_H __LIBRETRO__ PRIVATE _CRT_SECURE_NO_WARNINGS)
add_library(blipper STATIC ${CORE}/libretro/blipper.c)
target_include_directories(blipper PUBLIC ${CORE}/libretro)
set(WIZPLAYER_SNES9X_SOURCE "" CACHE PATH "Snes9x source for local builds")
include(cmake/bugfix_patches.cmake)
if(ANDROID)
  set_target_properties(binjnes gambatte blipper PROPERTIES POSITION_INDEPENDENT_CODE ON)
  # Match release performance in the FC emulation loop during Android testing.
  target_compile_options(binjnes PRIVATE "$<$<CONFIG:Debug>:-O3>")
  target_compile_definitions(binjnes PRIVATE "$<$<CONFIG:Debug>:NDEBUG>")
  add_library(gbwiz SHARED src/android_jni.cpp src/nes/core_bridge.c)
  target_include_directories(gbwiz PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  set_target_properties(gbwiz PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
  target_link_libraries(gbwiz PRIVATE gambatte blipper binjnes)
  if(WIZPLAYER_SNES9X_SOURCE)
    include(cmake/sfc.cmake)
    target_link_libraries(gbwiz PRIVATE wizplayer_sfc)
    target_compile_definitions(gbwiz PRIVATE WIZPLAYER_SFC)
  endif()
  return()
endif()
add_executable(WizPlayer src/main.cpp src/nes/windows.cpp src/nes/core_bridge.c)
target_include_directories(WizPlayer PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
set_target_properties(WizPlayer PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
set(SDL2_DIR ${CMAKE_CURRENT_SOURCE_DIR}/third_party/SDL2-2.32.10/cmake)
find_package(SDL2 CONFIG REQUIRED)
target_link_libraries(WizPlayer PRIVATE gambatte blipper binjnes SDL2::SDL2)
add_custom_command(TARGET WizPlayer POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL2::SDL2> $<TARGET_FILE_DIR:WizPlayer>)
target_link_libraries(WizPlayer PRIVATE user32 gdi32 comdlg32 shell32)
if(WIZPLAYER_SNES9X_SOURCE)
  include(cmake/sfc.cmake)
  target_link_libraries(WizPlayer PRIVATE wizplayer_sfc)
  target_compile_definitions(WizPlayer PRIVATE WIZPLAYER_SFC)
endif()
